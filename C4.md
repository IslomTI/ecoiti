### Шаг 2.1: Скрипт `deploy_project_01.sh`

Скрипт создает ВМ и назначает Floating IP.

Создайте файл `nano deploy_project_01.sh`:

```bash
#!/bin/bash

# === ПЕРЕМЕННЫЕ (ВСТАВЬТЕ СВОИ ID!) ===
IMAGE_ID="<ID_ОБРАЗА_alt-p11-cloud>"
FLAVOR_ID="<ID_FLAVOR_1vCPU_1RAM_10DISK>"
NET_ID="<ID_СЕТИ_Cloud-Network>"
KEY_NAME="<ИМЯ_КЛЮЧА_В_OPENSTACK>" 
SEC_GROUP="default" # Убедитесь, что открыты порты 22, 80, 443

echo "=== Запуск развертывания Project_01 ==="

# 1. Создание HAProxy
echo "Создание haproxy01..."
openstack server create --image $IMAGE_ID --flavor $FLAVOR_ID --nic net-id=$NET_ID --key-name $KEY_NAME --security-group $SEC_GROUP --wait haproxy01

# 2. Создание Game-серверов
for i in 01 02 03; do
    echo "Создание game$i..."
    openstack server create --image $IMAGE_ID --flavor $FLAVOR_ID --nic net-id=$NET_ID --key-name $KEY_NAME --security-group $SEC_GROUP --wait game$i
done

# 3. Назначение Floating IP для HAProxy
echo "Назначение Floating IP..."
FIP=$(openstack floating ip create public -f value -c floating_ip_address)
openstack server add floating ip haproxy01 $FIP

echo "Инфраструктура готова."
echo "HAProxy External IP: $FIP"

```

Сделайте исполняемым: `chmod +x deploy_project_01.sh`.

### Шаг 2.2: Скрипт `configure_project_01.sh`

Скрипт настраивает Docker, приложение и HAProxy.

Создайте файл `nano configure_project_01.sh`:

```bash
#!/bin/bash

# Получение IP адресов (Internal)
HA_IP=$(openstack server show haproxy01 -f value -c addresses | grep -oP '(?<=internal=)[0-9.]+')
G1_IP=$(openstack server show game01 -f value -c addresses | grep -oP '(?<=internal=)[0-9.]+')
G2_IP=$(openstack server show game02 -f value -c addresses | grep -oP '(?<=internal=)[0-9.]+')
G3_IP=$(openstack server show game03 -f value -c addresses | grep -oP '(?<=internal=)[0-9.]+')

SSH_OPTS="-o StrictHostKeyChecking=no -o UserKnownHostsFile=/dev/null -i /home/altlinux/.ssh/id_rsa"

echo "=== Настройка серверов приложений ==="

# Функция настройки одной ноды
setup_node() {
    NODE_IP=$1
    NAME=$2
    echo "Настройка $NAME ($NODE_IP)..."
    
    ssh $SSH_OPTS altlinux@$NODE_IP << EOF
        sudo apt-get update
        sudo apt-get install -y docker-ce unzip wget
        sudo systemctl enable --now docker
        
        # Скачивание приложения
        mkdir -p /home/altlinux/app
        # Ссылка из задания [286]
        wget "https://disk.yandex.ru/d/uhpN6U6UYRK_Nw" -O app_archive.zip
        # Примечание: Если wget не качает с Я.Диска напрямую, используйте заранее скачанный файл и scp
        # Но для скрипта предполагаем наличие ссылки или scp с jump host
        
        # (ОПЦИЯ) Если wget с Я.Диска не сработал, раскомментируйте scp ниже и закомментируйте wget выше
        # exit
EOF
    # scp $SSH_OPTS /path/to/local/app.zip altlinux@$NODE_IP:/home/altlinux/app/app_archive.zip
    
    ssh $SSH_OPTS altlinux@$NODE_IP << EOF
        cd /home/altlinux/app
        unzip app_archive.zip
        
        # Создание Dockerfile (Мультистейджинг)
        cat << DOCKER > Dockerfile
FROM python:3.9-alpine as builder
WORKDIR /app
COPY . .
# Тут могла быть сборка, но для статики или простого питона:
FROM python:3.9-alpine
WORKDIR /app
COPY --from=builder /app /app
CMD ["python3", "-m", "http.server", "80"] 
# (Замените CMD на реальный запуск приложения из README, если он отличается)
DOCKER

        # Сборка и запуск
        sudo docker build -t game-app .
        sudo docker run -d -p 80:80 --restart always --name game game-app
EOF
}

setup_node $G1_IP "game01"
setup_node $G2_IP "game02"
setup_node $G3_IP "game03"

echo "=== Настройка HAProxy ==="
ssh $SSH_OPTS altlinux@$HA_IP << EOF
    sudo apt-get update
    sudo apt-get install -y haproxy openssl

    # SSL Сертификат для game.au.team [277]
    sudo openssl req -x509 -nodes -days 365 -newkey rsa:2048 \
        -keyout /etc/ssl/private/game.key \
        -out /etc/ssl/certs/game.crt \
        -subj "/C=RU/ST=Moscow/L=Moscow/O=Skills/CN=game.au.team"
    
    cat /etc/ssl/certs/game.crt /etc/ssl/private/game.key | sudo tee /etc/ssl/private/game.pem

    # Конфигурация HAProxy [273-276]
    cat << CFG | sudo tee /etc/haproxy/haproxy.cfg
global
    log /dev/log local0
    log /dev/log local1 notice
    chroot /var/lib/haproxy
    stats socket /run/haproxy/admin.sock mode 660 level admin expose-fd listeners
    stats timeout 30s
    user haproxy
    group haproxy
    daemon

defaults
    log global
    mode http
    option httplog
    option dontlognull
    timeout connect 5000
    timeout client  50000
    timeout server  50000

# Статистика [274]
listen stats
    bind *:8404
    stats enable
    stats uri /haproxy?stats
    stats refresh 10s

# Frontend HTTPS [276]
frontend https_front
    bind *:443 ssl crt /etc/ssl/private/game.pem
    default_backend game_backend

# Backend
backend game_backend
    balance roundrobin
    server game01 $G1_IP:80 check
    server game02 $G2_IP:80 check
    server game03 $G3_IP:80 check
CFG

    sudo systemctl restart haproxy
EOF

# Добавляем запись в hosts на Cloud-ADM [275]
echo "\$HA_IP game01.dev.au.team game02.dev.au.team game03.dev.au.team game.au.team haproxy01.dev.au.team" | sudo tee -a /etc/hosts

```

Сделайте исполняемым: `chmod +x configure_project_01.sh`.

### Шаг 4.1: Инициализация кластера

1. **На master01:**
```bash
ssh altlinux@<IP_MASTER>
sudo apt-get install -y docker-ce && sudo systemctl enable --now docker

# Инициализация
sudo docker swarm init
# Скопируйте вывод команды (docker swarm join --token ...)

```


2. **На worker01 и worker02:**
```bash
# Установите docker
sudo apt-get install -y docker-ce && sudo systemctl enable --now docker
# Выполните команду join
sudo docker swarm join --token <ВАШ_ТОКЕН> <IP_MASTER>:2377

```


3. 
**Настройка Master (Drain):** 
На `master01`:


```bash
sudo docker node update --availability drain master01

```



Шаг 4.2: Локальный Registry 

1. **Запуск сервиса Registry:**
На `master01`:
```bash
sudo docker service create --name registry --publish published=5000,target=5000 registry:2

```


2. **Настройка доступа (на ВСЕХ нодах - master и workers):**
Чтобы Docker доверял http-регистру, отредактируйте `/etc/docker/daemon.json`:
```json
{
  "insecure-registries": ["<IP_MASTER01>:5000"]
}

```


Перезапустите Docker на всех нодах: `sudo systemctl restart docker`.

Шаг 4.3: Развертывание стека (Приложение) 

На `master01` создайте файл `docker-compose.yml`.


Ссылка на приложение:  ([https://disk.yandex.ru/d/ItDZh8DTdK5kAA](https://disk.yandex.ru/d/ItDZh8DTdK5kAA)).

```yaml
version: '3.8'

services:
  web:
    image: nginx:latest # Или образ, собранный из исходников и залитый в localhost:5000

      - "80:80"
      - "443:443"
    deploy:
      replicas: 2
      placement:
        constraints:
          - node.role == worker
    # Для HTTPS (задание 319) можно смонтировать сертификаты, если они есть

    image: redis:alpine

      replicas: 1
      placement:
        constraints:
           - node.role == worker

    image: postgres:13

      POSTGRES_PASSWORD: example
    deploy:
      replicas: 1

        constraints:
```
